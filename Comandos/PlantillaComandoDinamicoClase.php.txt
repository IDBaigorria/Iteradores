// Úsela cuando necesite crear un comando en tiempo de ejecución
// sin definir una clase en un archivo aparte.
// La clase anónima implementa la interfaz Comando y se registra
// a través de RegistroGlobal, que decidirá si puede registrarlo
// de inmediato o debe encolarlo para el próximo procesamiento.

use Iteradores\Comandos\Comando;
use Iteradores\Controlador\RegistroGlobal;

$comando = new class implements Comando {
    public static function nombre(): string
    {
        return 'dinamico:accion';
    }

    public static function solo_desarrollo(): bool
    {
        return false; // true si solo debe existir en desarrollo
    }

    public static function descripcion(): string
    {
        return 'Descripción breve del comando dinámico.';
    }

    public static function parametros(): array
    {
        return [
            // Definir aquí los parámetros igual que en una clase estática
        ];
    }

    public static function ejemplos(): array
    {
        return [
            'dinamico:accion --bandera1',
        ];
    }

    public function ejecutar(string $token, array $args): mixed
    {
        // Acceder a los argumentos ya parseados
        // $banderas = $args['banderas'];
        // $opciones = $args['opciones'];
        // $posicionales = $args['posicionales'];

        $controlador = RegistroGlobal::controlador();
        if ($controlador) {
            $controlador::escribir_salida("Ejecutando comando dinámico\n");
        }
        return true;
    }

    public function reversa(): ?callable
    {
        return null; // o devolver un callable
    }
};

// ─── REGISTRO AUTOMÁTICO ─────────────────────────────────
// RegistroGlobal decidirá la vía adecuada:
// - Si el Controlador ya está inicializado, lo registra inmediatamente.
// - Si no, lo encola para que se registre cuando el Controlador se inicie.
$controlador = RegistroGlobal::controlador();
if ($controlador) {
    // El Controlador ya existe: registrar directamente
    $controlador::registrar_comando_desde_instancia($comando);
} else {
    // El Controlador aún no está listo: encolar para registro diferido
    RegistroGlobal::encolar_comando($comando);
}
